213: snapshot_dependencies(S) ->
214:     case gen_server:reqids_size(maps:get(follow_requests,S)) of
215:         0 ->
216:             Requests = maps:fold(fun(Pid,Ns,Acc) ->
217:                 gen_server:send_request(Pid,get_stats,{Pid,Ns},Acc)
218:             end,maps:get(snapshots,S),maps:get(owners,S)),
219:             resume_projection(S#{snapshots := Requests});
220:         _ -> {ok,S}
221:     end.

223: %% Keep pending registrations until their reply even after view cancellation:
224: %% only then is the owner-issued reference known and can be released. At most
225: %% one opening batch is retained; a new open waits for this cleanup notice.
226: follow_response(Message,S = #{follow_requests := Requests}) ->
227:     case gen_server:check_response(Message,Requests,true) of
228:         {Reply,{Pid,Ns,Anchor,Height},Remaining} ->
229:             Next = S#{follow_requests := Remaining},

231:                 {true,{reply,{ok,Ref}},Pid} ->

233:                         (maps:get(follows,S))#{Ref => {Ns,Anchor,Height,false}}});
234:                 {Active,_,_} ->
235:                     case Reply of {reply,{ok,Ref}} -> unfollow(Ref); _ -> ok end,
236:                     case Active of
237:                         true -> reply(#{type => unavailable,reason => dependency_unavailable},release(Next));
238:                         false -> finish_follow_cleanup(Next)
239:                     end
240:             end;
241:         _ -> snapshot_response(Message,S)
242:     end;
243: follow_response(Message,S) -> snapshot_response(Message,S).
244: 
245: finish_follow_cleanup(S = #{follow_requests := Requests}) ->
246:     case gen_server:reqids_size(Requests) of

254:          not maps:is_key(opening,S) andalso not maps:is_key(follow_requests,S) of
255:         true -> refresh(S,dependency_recovered);

259: unfollow(Ref) ->

261:         {ok, Request} -> _ = gen_server:receive_response(Request,0), ok;

265: snapshot_response(Message,S = #{snapshots := Requests}) ->
266:     case gen_server:check_response(Message,Requests,true) of

279:         _ -> {ok,S}

283: resume_projection(S = #{pending := {Digest, Expiry, Outcome}}) ->
284:     case {Expiry > quod_time:now_ms() andalso candidate_live(S),dependencies_ready(S)} of
285:         {false,_} -> reply(#{type => unavailable,reason => expired},release(S));
286:         {true,true} -> open_view(Outcome,Digest,Expiry,maps:remove(pending,S));
287:         {true,false} -> {ok,S}

291: dependencies_ready(S) ->
292:     gen_server:reqids_size(maps:get(follow_requests,S)) =:= 0 andalso
293:     gen_server:reqids_size(maps:get(snapshots,S)) =:= 0 andalso

295:     lists:all(fun({_,_,_,R}) -> R end,maps:values(maps:get(follows,S))).

339: attach_current(Pid,Expected,Marks,Digest,Height,Expiry,S) ->
340:     M = case maps:find(monitor,S) of {ok,Existing} -> Existing; error -> monitor(process,Pid) end,

381: projection_deadline(S,Expiry) ->

387: %% The ontology's subject and the body identity remain separate. Duplicate
388: %% occurrences may depict one body, but every occurrence has exactly one source.
389: body_bindings(Marks, Control, #{poses := Poses}) ->
390:     try
391:         Available = maps:from_list([{{Id, R}, true} || {Id, R, world, _, _, _} <- Poses]),
392:         Rows = [{Mark, mark_body(Pose, Depicts)} ||

394:         true = length(Rows) =:= length(Marks),
395:         true = map_size(maps:from_list(Rows)) =:= length(Rows),
396:         true = lists:all(fun({Mark, Body}) -> is_binary(Mark) andalso
397:                             (Body =:= none orelse maps:is_key(Body, Available)) end, Rows),
398:         Controlled = case Control of
399:             none -> [];

401:                 true = maps:is_key({Id,Revision},Available), [{{Id,Revision},true}]
402:         end,
403:         {ok, maps:from_list(Controlled ++ [{Body, true} || {_, Body} <- Rows, Body =/= none])}
404:     catch error:_ -> error end.

301: install(Spec = #{bodies := Bodies}, S = #s{native = Resource}) ->
302:     Hinges = maps:get(hinges,Spec,[]),
303:     Geometry = maps:with([bodies,hinges,restores],Spec),
304:     Result = case S#s.installed_geometry =:= Geometry of
305:         true -> {ok,0};

309:                 {ok,J} ->

311:                                       maps:get(Id,S#s.revisions,none) =/= Rev],
312:                     case quod_physics_native:restore_rest(Resource,Restores) of
313:                         {ok,{}} -> {ok,N+J}; Error -> Error end;
314:                 Error -> Error end;
315:             Error -> Error
316:         end
317:         end)
318:     end,
319:     case Result of
320:         {ok, Changes} ->
321:             {ControlChanges,DoorsInstalled} = install_controls(S#s{spec = Spec,installed_geometry = Geometry,
322:                 revisions = maps:from_list([{Id,Rev} || {Id,Rev,_,_,_,_} <- Bodies])}),
323:             AgentControls = maps:get(agent_controls, Spec, []),
324:             ControlChanged = S#s.agent_controls =/= AgentControls,
325:             Installed = #{fence => fence(Spec),incarnation => S#s.token,structure => maps:get(structure,Spec)},
326:             Inputs = maps:filtermap(fun(_,Row = #{control := Control,expected := Expected}) ->
327:                 case retained_binding(Control,maps:get(agent_bindings,Spec,[])) of
328:                     false -> false;
329:                     true ->
330:                         Retained = case lists:member(Control,AgentControls) andalso current_projection(Installed,Expected) of
331:                             true -> Row;
332:                             false -> (suspend_input(Row))#{expected => maps:merge(Expected,Installed)}
333:                         end,
334:                         %% Changed physical support wakes grounded mechanics,
335:                         %% even when the previous frame was fully settled.
336:                         {true,Retained#{falling => maps:get(falling,Retained) orelse
337:                             (Changes > 0 andalso grounded_control(Control))}}
338: 
339:                 end
340:             end,S#s.inputs),
341:             Guard = maps:get(navigation_guard,Spec),
342:             GuardChanged = Guard =/= S#s.installed_guard,
343:             Controlled = reconcile_controller(DoorsInstalled#s{agent_controls = AgentControls,inputs = Inputs,installed_guard = Guard}),

353:                     end;
354:                 _ -> notify(installed,do_step(Controlled#s{installs = S#s.installs +
355:                     case Changes > 0 of true -> 1; false -> 0 end}))
356:             end;
357:         {error,Reason} -> fail(Reason,S)
358:     end.

376: install_controls(S = #s{spec = Spec, controls = Old, native = Resource, observations = Observations}) ->
377:     Activities = maps:get(activities,Spec,#{}),
378:     Doors = maps:filter(fun(K,_) -> is_integer(K) end,Activities),
379:     Desired = maps:map(fun(D,#{state := State}) ->
380:         Revision = maps:get(D,S#s.revisions),
381:         case State of {closing,R} -> {Revision,R}; _ -> stopped end
382:     end,Doors),
383:     Kept = maps:filter(fun(D,#{request := R}) ->
384:         case maps:get(D,Activities,none) of #{request := R} -> true; _ -> false end
385:     end,Observations),
386:     maps:fold(fun(D,Control,{Count,Acc}) ->
387:         case maps:get(D,Old,stopped) =:= Control of
388:             true -> {Count,Acc};
389:             false ->
390:                 Drive = case {Control,maps:get(D,Doors)} of
391:                     {{_,_},#{custody := none}} -> true;
392:                     _ -> false

395:                     true ->

399:                         end;
400:                     false -> {Count,Acc}
401:                 end
402:         end
403:     end,{0,S#s{controls = Desired,observations = Kept}},Desired).
