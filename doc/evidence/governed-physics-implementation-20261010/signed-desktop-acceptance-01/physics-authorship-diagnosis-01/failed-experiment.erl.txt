authored_physics_library_single_transaction_test_() ->
    {timeout, 90, fun() ->
        Dir = filename:join("/tmp", "quod-physics-authorship-" ++
                            integer_to_list(erlang:unique_integer([positive]))),
        Ctx = start(Dir),
        try
            {ontology_ref, Ns, Anchor} = maps:get(lobby, Ctx),
            {ok, AgentSource} = file:read_file(filename:join(code:priv_dir(quod), "ontologies/agent_instance.pl")),
            {ok, PhysicsSource} = file:read_file(filename:join(code:priv_dir(quod), "ontologies/quod_physics.pl")),
            ColdName = <<"physics_authorship_", (binary:encode_hex(crypto:strong_rand_bytes(8)))/binary>>,
            ?assertError(badarg, binary_to_existing_atom(ColdName, utf8)),
            Source = <<AgentSource/binary, "\n", PhysicsSource/binary, "\n", ColdName/binary, "(ready).\n">>,
            Install = {transaction, {findall, ok, editor_conjunction([
                editor_scope(Ns, {'$quod_predicate_source', {'/', agent_hosted, 4}, <<>>}),
                editor_scope(Ns, {'$quod_predicate_source', {'/', physics_enabled, 2}, <<>>}),
                {'$quod_program_source', {0}, Source},
                {findall, ok, {',', {member, {1}, {0}},
                                  editor_scope(Ns, {assertz, {1}})}, {2}},
                {length, {0}, {3}}, {length, {2}, {3}}]), [ok]}},
            Before = quod_prolog:applied(Ns),
            {Bytes, Signature} = signed_request(Ctx, execute, Install),
            {ok, Evidence = #{request := #{signing_public_key := SigningKey}}} =
                quod_client_goal_target:verify_request(Bytes, Signature),
            {ok, Peer} = application:get_env(quod, node_pubkey),
            Owner = {session, crypto:strong_rand_bytes(32), SigningKey},
            {ok, {Evidence, Materialized, Principal, Owner}} =
                quod_client_goal_target:prepare_local(Evidence, Peer, Owner, none),
            case quod_prolog:execute_signed(Evidence, Materialized, Principal) of
                {ok, _, _} -> ok;
                Other -> error({source_program_authorship_result, Other})
            end,
            ?assertEqual(Before + 1, quod_prolog:applied(Ns)),
            ?assertEqual(Anchor, quod_simplex:genesis_hash(Ns)),
            Cold = binary_to_existing_atom(ColdName, utf8),
            ?assertMatch({ok, [#{}], _}, read(Ctx, editor_scope(Ns, {Cold, ready}))),
            ?assertEqual(#{}, gen_server:call(quod_reg:via({quod_runtime, Ns}), physics_worlds))
        after stop(Ctx), file:del_dir_r(Dir) end
    end}.
