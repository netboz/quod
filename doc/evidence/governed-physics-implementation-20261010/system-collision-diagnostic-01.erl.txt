-module('system-collision-diagnostic-01').
-export([main/0]).

%% Deliberately exercises a pre-upgrade interpreted name in retained genesis.
%% Exit 1 means the common registration silently displaced retained content.
main() ->
    {ok,_} = application:ensure_all_started(gproc),
    Ns = <<"projection-collision-diagnostic">>, Anchor = <<0:256>>,
    Fact = {physics_body_state,original_space,original_body,retained_value},
    Genesis = quod_simplex:test_genesis_tx(
        #{node_id => Anchor,mode => create,committee => [],
          external_predicate_modules => [],
          genesis_diff => quod_ct:diff_for(Fact)},Ns,Anchor,<<97:256>>),
    {ok,Outcomes} = quod_outcome:open(Ns,Anchor,#{outcome_backend => memory}),
    Base = quod_committed_projection:new_est(),
    Projection = quod_committed_projection:new({Ns,Anchor},0,Base,Outcomes,none),
    Entry = quod_ct:committed_entry(<<"projection:codec-only">>,1,{batch,[Genesis]}),
    case quod_committed_projection:apply_entry(Entry,1,Projection) of
        {ok,P,_} ->
            Descriptor = quod_predicates:descriptor(quod_committed_projection:est(P),
                                                    {physics_body_state,3}),
            io:format("retained_genesis_accepted=true descriptor=~p~n",[Descriptor]),
            io:format("STOP: retained interpreted definition displaced by shared registration~n"),
            halt(1);
        {error,Reason} ->
            io:format("retained_genesis_refused=~p~n",[Reason]), halt(0)
    end.
